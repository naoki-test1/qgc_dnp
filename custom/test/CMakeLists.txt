cmake_minimum_required(VERSION 3.22)
project(PhotoCaptureFeedbackTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
enable_testing()
add_executable(PhotoCaptureFeedbackTest PhotoCaptureFeedbackTest.cc ../src/PhotoCaptureFeedback.cc)
target_include_directories(PhotoCaptureFeedbackTest PRIVATE ../src)
target_link_libraries(PhotoCaptureFeedbackTest PRIVATE Qt6::Core Qt6::Test)
add_test(NAME PhotoCaptureFeedbackTest COMMAND PhotoCaptureFeedbackTest)

find_package(Qt6 QUIET COMPONENTS QuickTest Quick)
if(TARGET Qt6::QuickTest)
    add_executable(PhotoCapturePreviewTest PhotoCapturePreviewTest.cc ../src/PhotoPreviewCapture.cc)
    target_include_directories(PhotoCapturePreviewTest PRIVATE ../src)
    target_link_libraries(PhotoCapturePreviewTest PRIVATE Qt6::QuickTest Qt6::Quick)
    target_compile_definitions(PhotoCapturePreviewTest PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/qml")
    add_test(NAME PhotoCapturePreviewTest COMMAND PhotoCapturePreviewTest -import "${CMAKE_CURRENT_SOURCE_DIR}/imports")
    set_tests_properties(PhotoCapturePreviewTest PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
